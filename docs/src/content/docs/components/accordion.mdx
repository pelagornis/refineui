## Usage

```tsx
import { Accordion, AccordionItem, AccordionTrigger, AccordionContent } from "@refineui/react";

<Accordion type="single" collapsible>
  <AccordionItem value="item-1">
    <AccordionTrigger>What is RefineUI?</AccordionTrigger>
    <AccordionContent>
      RefineUI is a design system and component library.
    </AccordionContent>
  </AccordionItem>
  <AccordionItem value="item-2">
    <AccordionTrigger>How do I install it?</AccordionTrigger>
    <AccordionContent>
      Run: bun add @refineui/react @refineui/tokens
    </AccordionContent>
  </AccordionItem>
</Accordion>

<Accordion type="multiple" defaultValue={["item-1", "item-2"]}>
  {/* Multiple panels can be open */}
</Accordion>

<Accordion size="lg" value={openItem} onValueChange={setOpenItem}>
  {/* Controlled accordion */}
</Accordion>
```
