## Usage

```tsx
import { Avatar, AvatarImage, AvatarFallback, AvatarBadge } from "@refineui/react";

<Avatar>
  <AvatarImage src="/user.jpg" alt="User Name" />
  <AvatarFallback>UN</AvatarFallback>
</Avatar>

<Avatar size="lg" showStatus status="online">
  <AvatarImage src="/avatar.jpg" alt="User" />
  <AvatarFallback>U</AvatarFallback>
  <AvatarBadge status="online" />
</Avatar>

<Avatar layout="icon" color="blue">
  <AvatarIcon name="user" />
</Avatar>

<Avatar layout="initials" color="green">
  <AvatarText>AB</AvatarText>
</Avatar>
```
