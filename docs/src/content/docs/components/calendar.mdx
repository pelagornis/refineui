## Usage

```tsx
import { Calendar } from "@refineui/react";

<Calendar value={selectedDate} onChange={setSelectedDate} />

<Calendar
  mode="range"
  rangeStart={startDate}
  rangeEnd={endDate}
  onRangeChange={(start, end) => {
    setStartDate(start);
    setEndDate(end);
  }}
/>

<Calendar weekStartsOn={1} defaultMonth={new Date(2024, 0)} />
```
