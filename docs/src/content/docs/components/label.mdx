## Usage

```tsx
import { Label, LabelRequired } from "@refineui/react";

<Label htmlFor="email">Email address</Label>

<Label htmlFor="password" required>
  Password
</Label>

<Label size="lg" disabled>
  Disabled Label
</Label>

<Label htmlFor="name">
  Name <LabelRequired />
</Label>
```
