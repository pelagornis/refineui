## Usage

```tsx
import { Radio } from "@refineui/react";

<Radio name="plan" value="basic" label="Basic Plan" />

<Radio
  name="plan"
  value="pro"
  label="Pro Plan"
  description="Includes advanced features"
  checked={selectedPlan === "pro"}
  onChange={(e) => setSelectedPlan(e.target.value)}
/>

<Radio
  name="plan"
  value="enterprise"
  label="Enterprise"
  allowUncheck
/>
```
