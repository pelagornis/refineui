## Usage

```tsx
import { SearchField, SearchFieldIcon, SearchFieldInput, SearchFieldClear } from "@refineui/react";

<SearchField value={query} onValueChange={setQuery} />

<SearchField
  value={searchTerm}
  onValueChange={setSearchTerm}
  appearance="filled"
  aria-label="Search products"
/>

<SearchField appearance="plain" defaultValue="initial query" />
```
