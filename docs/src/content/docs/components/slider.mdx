## Usage

```tsx
import { Slider } from "@refineui/react";

<Slider min={0} max={100} value={volume} onChange={setVolume} />

<Slider
  min={0}
  max={10}
  step={1}
  value={rating}
  onChange={setRating}
  size="md"
/>
```
