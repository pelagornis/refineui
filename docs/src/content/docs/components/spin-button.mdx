## Usage

```tsx
import { SpinButton } from "@refineui/react";

<SpinButton value={quantity} onChange={setQuantity} />

<SpinButton
  value={amount}
  onChange={setAmount}
  min={0}
  max={100}
  step={5}
  placeholder="Enter amount"
  size="lg"
/>
```
