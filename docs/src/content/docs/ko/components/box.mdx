## 사용법

```tsx
import { Box } from "@refineui/react";

<Box padding="sizeLarge" background="backgroundSurface" radius="roundedLarge">
  Content
</Box>

<Box
  as="section"
  paddingX="sizeLarge"
  paddingY="sizeSmall"
  border="strokeWidthThin"
  borderColor="borderDefault"
  background="backgroundPrimary"
>
  Outlined section
</Box>
```
