## 사용법

```tsx
import { Container, Stack, Text } from "@refineui/react";

<Container padding="sizeXXLarge">
  <Stack gap="sizeLarge">
    <Text variant="headingMd" as="h1">Page title</Text>
    <Text variant="bodyMd">Body</Text>
  </Stack>
</Container>
```
