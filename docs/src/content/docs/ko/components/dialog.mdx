## 사용법

```tsx
import {
  Dialog,
  DialogTrigger,
  DialogContent,
  DialogHeader,
  DialogTitle,
  DialogDescription,
  DialogClose,
  Button,
} from "@refineui/react";

<Dialog>
  <DialogTrigger asChild>
    <Button>Open Dialog</Button>
  </DialogTrigger>
  <DialogContent>
    <DialogHeader>
      <DialogTitle>Confirm Action</DialogTitle>
      <DialogDescription>
        Are you sure you want to continue?
      </DialogDescription>
    </DialogHeader>
    <DialogClose asChild>
      <Button>Cancel</Button>
    </DialogClose>
  </DialogContent>
</Dialog>

<Dialog size="sm" open={isOpen} onOpenChange={setIsOpen}>
  <DialogContent scrollable={false}>
    {/* Content with custom scroll handling */}
  </DialogContent>
</Dialog>
```
