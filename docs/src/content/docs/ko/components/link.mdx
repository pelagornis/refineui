## 사용법

```tsx
import { Link } from "@refineui/react";

<Link href="/docs">Documentation</Link>

<Link href="https://example.com" target="_blank">
  External Link
</Link>

<Link href="https://example.com" showExternalIcon={false}>
  No Icon
</Link>

<Link href="/disabled" disabled>
  Disabled Link
</Link>
```
