## 사용법

```tsx
import { Popover, PopoverTrigger, PopoverContent, Button } from "@refineui/react";

<Popover>
  <PopoverTrigger>
    <Button>Show Info</Button>
  </PopoverTrigger>
  <PopoverContent>
    <p>Additional information here</p>
  </PopoverContent>
</Popover>

<Popover open={isOpen} onOpenChange={setIsOpen}>
  <PopoverTrigger>
    <Button>Toggle</Button>
  </PopoverTrigger>
  <PopoverContent placement="top" align="center">
    <p>Popover content</p>
  </PopoverContent>
</Popover>
```
