```tsx

<ProgressStepper value={currentStep} onValueChange={setCurrentStep}>

      <ProgressStepperMarker />
      <ProgressStepperLabel>Account</ProgressStepperLabel>
    </ProgressStepperItem>
    <ProgressStepperItem value={2}>
      <ProgressStepperMarker />

    <ProgressStepperItem value={3}>
```
