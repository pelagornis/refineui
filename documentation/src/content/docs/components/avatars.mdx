## Usage

```tsx
import { Avatars } from "@refineui/react";

<Avatars
  avatars={[
    { src: "/user1.jpg", alt: "User 1" },
    { src: "/user2.jpg", alt: "User 2" },
    { src: "/user3.jpg", alt: "User 3" }
  ]}
/>

<Avatars
  avatars={teamMembers}
  size="lg"
  max={3}
  layout="spread"
/>
```
