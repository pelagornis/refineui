Contextual text popup on hover or focus.

## Installation

```bash
bun add @refineui/react @refineui/tokens
```

## Usage

```tsx
import { Tooltip } from "@refineui/react";

<Tooltip trigger={<Button>Hover me</Button>} content="Helpful tooltip text" />

<Tooltip
  trigger={<IconButton icon="info" />}
  content="Extended information"
  position="Top"
  align="Center"
/>
```
